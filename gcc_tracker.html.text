const CITY_ORDER = ["Chennai", "Bengaluru", "Hyderabad", "Pune", "Mumbai/NCR"];

function fmtDate(iso) {
  try {
    const d = new Date(iso + "T00:00:00");
    return d.toLocaleDateString("en-IN", { day: "numeric", month: "short", year: "numeric" });
  } catch (e) { return iso; }
}

function renderStats() {
  const statsEl = document.getElementById("stats");
  const total = SIGNALS.length;
  const flagged = SIGNALS.filter(s => s.flagged).length;
  const cities = new Set(SIGNALS.map(s => s.city)).size;
  const sectors = new Set(SIGNALS.map(s => s.sector)).size;
  statsEl.innerHTML = `
    <div class="stat"><div class="n">${total}</div><div class="l">Total signals</div></div>
    <div class="stat"><div class="n">${flagged}</div><div class="l">Chennai signals</div></div>
    <div class="stat"><div class="n">${cities}</div><div class="l">Cities covered</div></div>
    <div class="stat"><div class="n">${sectors}</div><div class="l">Sectors</div></div>
  `;
}

function populateFilters() {
  const cityFilter = document.getElementById("cityFilter");
  const typeFilter = document.getElementById("typeFilter");
  const cities = [...new Set(SIGNALS.map(s => s.city))].sort(
    (a, b) => CITY_ORDER.indexOf(a) - CITY_ORDER.indexOf(b)
  );
  cities.forEach(c => {
    const opt = document.createElement("option");
    opt.value = c; opt.textContent = c;
    cityFilter.appendChild(opt);
  });
  const types = [...new Set(SIGNALS.map(s => s.signalType))].sort();
  types.forEach(t => {
    const opt = document.createElement("option");
    opt.value = t; opt.textContent = t;
    typeFilter.appendChild(opt);
  });
}

function cardHtml(s) {
  const sourceHtml = s.source
    ? `<a href="${s.source}" target="_blank" rel="noopener">${s.sourceLabel}</a>`
    : `<span class="no-source">${s.sourceLabel}</span>`;
  return `
    <div class="card ${s.flagged ? "flagged" : ""}">
      <div class="card-top">
        <span class="company">${s.company}</span>
        <span>
          <span class="chip">${s.signalType}</span>
          ${s.flagged ? '<span class="chip flag">Chennai</span>' : ""}
        </span>
      </div>
      <div class="meta-line">${s.city} · ${s.sector} · ${fmtDate(s.eventDate)}</div>
      <div class="notes">${s.notes}</div>
      <div class="source-line">${sourceHtml}</div>
    </div>
  `;
}

function render() {
  const q = document.getElementById("searchBox").value.trim().toLowerCase();
  const cityVal = document.getElementById("cityFilter").value;
  const typeVal = document.getElementById("typeFilter").value;

  let filtered = SIGNALS.filter(s => {
    if (cityVal && s.city !== cityVal) return false;
    if (typeVal && s.signalType !== typeVal) return false;
    if (q) {
      const hay = (s.company + " " + s.sector + " " + s.notes + " " + s.signalType).toLowerCase();
      if (!hay.includes(q)) return false;
    }
    return true;
  });

  filtered.sort((a, b) => new Date(b.eventDate) - new Date(a.eventDate));

  const resultsEl = document.getElementById("results");
  if (filtered.length === 0) {
    resultsEl.innerHTML = `<div class="empty-state">No signals match your filters.</div>`;
    return;
  }

  const groups = {};
  filtered.forEach(s => {
    groups[s.city] = groups[s.city] || [];
    groups[s.city].push(s);
  });

  const orderedCities = Object.keys(groups).sort(
    (a, b) => CITY_ORDER.indexOf(a) - CITY_ORDER.indexOf(b)
  );

  resultsEl.innerHTML = orderedCities.map(city => `
    <div class="city-group">
      <h2>${city} <span class="count">${groups[city].length} signal${groups[city].length > 1 ? "s" : ""}</span></h2>
      ${groups[city].map(cardHtml).join("")}
    </div>
  `).join("");
}

renderStats();
populateFilters();
render();

document.getElementById("searchBox").addEventListener("input", render);
document.getElementById("cityFilter").addEventListener("change", render);
document.getElementById("typeFilter").addEventListener("change", render);
